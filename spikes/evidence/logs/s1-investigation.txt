    37	export function resolveWritePath(target, root) {
    38	    const absTarget = path.resolve(target);
    39	    const absRoot = path.resolve(root);
    40	    if (!isPathWithin(absRoot, absTarget)) {
    41	        throw new Error(`Refusing to write outside rootfs: ${absTarget}`);
    42	    }
    43	    const rel = path.relative(absRoot, absTarget);
    44	    if (!rel || rel === ".") {
    45	        return absRoot;
    46	    }
    47	    const parts = rel.split(path.sep);
    48	    let current = absRoot;
    49	    for (let idx = 0; idx < parts.length; idx++) {
    50	        const next = path.join(current, parts[idx]);
    51	        try {
    52	            const stat = fs.lstatSync(next);
    53	            if (!stat.isSymbolicLink()) {
    54	                current = next;
    55	                continue;
    56	            }
    57	            const linkTarget = fs.readlinkSync(next);
    58	            const resolved = path.resolve(path.dirname(next), linkTarget);
    59	            if (!isPathWithin(absRoot, resolved)) {
    60	                throw new Error(`Refusing to write through symlinked path: ${absTarget}`);
    61	            }
    62	            current = resolved;
    63	        }
    64	        catch (err) {
    65	            if (err?.code === "ENOENT") {
    66	                return path.join(current, ...parts.slice(idx));
    67	            }
    68	            throw err;
    69	        }
    70	    }
    71	    return current;
